// database
CREATE DATABASE dogalimDB
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

// users & address
CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_name VARCHAR(255) NOT NULL,
  mail_adr VARCHAR(255) UNIQUE NOT NULL,
  password VARCHAR(255) NOT NULL,
  defaultAddressId INT NULL
);

CREATE TABLE address (
  id INT AUTO_INCREMENT PRIMARY KEY,
  userId INT NOT NULL,
  phone VARCHAR(255) NOT NULL,
  city VARCHAR(255) NOT NULL,
  district VARCHAR(255) NOT NULL,
  postalCode VARCHAR(255) NULL,
  addressLine TEXT NOT NULL,
  createdAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_address_user
    FOREIGN KEY (userId)
    REFERENCES users(id)
    ON DELETE CASCADE
);

ALTER TABLE users
ADD CONSTRAINT fk_user_default_address
FOREIGN KEY (defaultAddressId)
REFERENCES address(id)
ON DELETE SET NULL;

// product
CREATE TABLE product (
  id INT AUTO_INCREMENT PRIMARY KEY,

  name VARCHAR(255) NOT NULL,
  price INT NOT NULL,

  ratingSum INT DEFAULT 0,
  ratingCount INT DEFAULT 0,

  explanation TEXT NOT NULL,

  sellerId INT
);

// cart
CREATE TABLE cart (
  id INT AUTO_INCREMENT PRIMARY KEY,
  userId INT,
  CONSTRAINT fk_cart_user
    FOREIGN KEY (userId)
    REFERENCES users(id)
    ON DELETE CASCADE
);

CREATE TABLE cart_item (
  id INT AUTO_INCREMENT PRIMARY KEY,

  cartId INT NOT NULL,
  productId INT NOT NULL,

  quantity INT DEFAULT 1,

  CONSTRAINT fk_cartitem_cart
    FOREIGN KEY (cartId)
    REFERENCES cart(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_cartitem_product
    FOREIGN KEY (productId)
    REFERENCES product(id)
    ON DELETE CASCADE,

  UNIQUE KEY uq_cart_product (cartId, productId)
);

// prodht
CREATE TABLE prodht (
  id INT AUTO_INCREMENT PRIMARY KEY,
  hashtag VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE product_prodht (
  productId INT NOT NULL,
  prodhtId INT NOT NULL,

  PRIMARY KEY (productId, prodhtId),

  CONSTRAINT fk_pp_product
    FOREIGN KEY (productId)
    REFERENCES product(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_pp_prodht
    FOREIGN KEY (prodhtId)
    REFERENCES prodht(id)
    ON DELETE CASCADE
);

// p-comm
CREATE TABLE pcomm (
  id INT AUTO_INCREMENT PRIMARY KEY,

  productId INT NOT NULL,
  commenterId INT NOT NULL,

  comment TEXT NOT NULL,
  evaluation INT NOT NULL,

  createdAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  CONSTRAINT fk_pcomm_product
    FOREIGN KEY (productId)
    REFERENCES product(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_pcomm_user
    FOREIGN KEY (commenterId)
    REFERENCES users(id)
    ON DELETE CASCADE,

  UNIQUE KEY uq_product_commenter (productId, commenterId)
);

// liked-prod
CREATE TABLE liked_products (
  id INT AUTO_INCREMENT PRIMARY KEY,

  user_id INT NOT NULL,
  product_id INT NOT NULL,

  likedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  CONSTRAINT fk_liked_user
    FOREIGN KEY (user_id)
    REFERENCES users(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_liked_product
    FOREIGN KEY (product_id)
    REFERENCES product(id)
    ON DELETE CASCADE,

  UNIQUE KEY uq_user_product (user_id, product_id)
);

// sellers
CREATE TABLE sellers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_name VARCHAR(255) NOT NULL,
    mail_adr VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    explanation TEXT NOT NULL,
    ratingSum INT NOT NULL DEFAULT 0,
    ratingCount INT NOT NULL DEFAULT 0
);

// s-comm
CREATE TABLE s_comm (
  id INT AUTO_INCREMENT PRIMARY KEY,

  sellerId INT,
  commenterId INT,

  comment TEXT NOT NULL,
  evaluation INT NOT NULL,

  createdAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE KEY unique_comment (commenterId, sellerId),

  CONSTRAINT fk_scomm_seller
    FOREIGN KEY (sellerId)
    REFERENCES sellers(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_scomm_user
    FOREIGN KEY (commenterId)
    REFERENCES users(id)
    ON DELETE CASCADE
);

// refresh-tokens-users
CREATE TABLE refresh_tokens (
  id INT AUTO_INCREMENT PRIMARY KEY,

  refreshToken VARCHAR(64) NOT NULL,
  userId INT NOT NULL,

  deviceid VARCHAR(128) NOT NULL,
  ipadress VARCHAR(255) NULL,
  useragent TEXT NULL,

  expiresat TIMESTAMP NOT NULL,

  revoked BOOLEAN DEFAULT FALSE,
  revokedat TIMESTAMP NULL,

  UNIQUE KEY uniq_refresh_token (refreshToken),

  CONSTRAINT fk_refresh_user
    FOREIGN KEY (userId)
    REFERENCES users(id)
    ON DELETE CASCADE
);

// refresh-tokens-seller
CREATE TABLE seller_refresh_tokens (
  id INT AUTO_INCREMENT PRIMARY KEY,

  refreshToken VARCHAR(64) NOT NULL,
  sellerId INT NOT NULL,

  deviceid VARCHAR(128) NOT NULL,
  ipadress VARCHAR(255) NULL,
  selleragent TEXT NULL,

  expiresat TIMESTAMP NOT NULL,

  revoked BOOLEAN DEFAULT FALSE,
  revokedat TIMESTAMP NULL,

  UNIQUE KEY uniq_seller_refresh_token (refreshToken),

  CONSTRAINT fk_refresh_seller
    FOREIGN KEY (sellerId)
    REFERENCES sellers(id)
    ON DELETE CASCADE
);

// hashtags-seller
CREATE TABLE hashtags (
  id INT AUTO_INCREMENT PRIMARY KEY,
  hashtag VARCHAR(255) NOT NULL
);

// hashtag-seller-join-table
CREATE TABLE sellers_hashtags_hashtags (
  sellersId INT NOT NULL,
  hashtagsId INT NOT NULL,

  PRIMARY KEY (sellersId, hashtagsId),

  CONSTRAINT fk_seller_hashtag_seller
    FOREIGN KEY (sellersId)
    REFERENCES sellers(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_seller_hashtag_hashtag
    FOREIGN KEY (hashtagsId)
    REFERENCES hashtags(id)
    ON DELETE CASCADE
);

// follows
CREATE TABLE follows (
  id INT AUTO_INCREMENT PRIMARY KEY,

  user_id INT NOT NULL,
  seller_id INT NOT NULL,

  followedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE KEY uq_user_seller (user_id, seller_id),

  INDEX idx_user (user_id),
  INDEX idx_seller (seller_id),

  CONSTRAINT fk_follow_user
    FOREIGN KEY (user_id)
    REFERENCES users(id)
    ON DELETE CASCADE,

  CONSTRAINT fk_follow_seller
    FOREIGN KEY (seller_id)
    REFERENCES sellers(id)
    ON DELETE CASCADE
);
